tryda Ostrava {
    rynek(chachar[] vstup) {
        kaj (vstup == chuj) {
            Konzola.pravit("Chybi vstupni soubor") pyco
            davaj pyco
        }

        toz Dryst menoSouboru = zrob Dryst(vstup) pyco
        toz Dekoder dekoder = zrob Dekoder(3) pyco
        toz Dryst tajnyDryst = Dekoder.dekoduj(menoSouboru) pyco
        Konzola.pravit(tajnyDryst) pyco
    }
} 

tryda Dekoder {
    toz cyslo posun pyco

    Dekoder(cyslo posun) {
        joch.posun = posun pyco
    }

    Dryst dekoduj(Dryst menoSouboru) {
        toz Dryst dryst = "" pyco

        toz Citac citac = zrob Citac() pyco
        citac.otevr(menoSouboru) pyco

        toz Dryst radka = citac.citajRadku() pyco

        rubat (radka != chuj) {
            toz Dryst[] cisla = radka.rozdel(' ') pyco
            toz cyslo soucet = 0 pyco
            toz cyslo synek = 0 pyco

            rubat (synek < Pole.velikost(cisla)) {
                soucet = soucet + CysloCele.preved(cisla[synek]) pyco
                synek = synek + 1 pyco
            }

            toz chachar znak = soucet + joch.posun pyco
            dryst = dryst.pridaj(znak) pyco
            radka = citac.citajRadku() pyco
        }

        davaj dryst pyco
    }
}
